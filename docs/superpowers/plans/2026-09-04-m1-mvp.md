```python
# app/providers/llm/openai_compat.py
import json
from typing import AsyncIterator

import httpx


class OpenAICompatLLM:
    def __init__(self, base_url: str, model: str, api_key: str | None = None, timeout: float = 120.0):
        self.base_url = base_url.rstrip("/")
        self.model = model
        self.api_key = api_key
        self.timeout = timeout

    async def stream_chat(self, messages: list[dict], **params) -> AsyncIterator[str]:
        headers = {"Authorization": f"Bearer {self.api_key}"} if self.api_key else {}
        payload = {"model": self.model, "messages": messages, "stream": True, **params}
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            async with client.stream("POST", f"{self.base_url}/chat/completions",
                                     json=payload, headers=headers) as resp:
                resp.raise_for_status()
                async for line in resp.aiter_lines():
                    if not line.startswith("data: "):
                        continue
                    data = line[6:]
                    if data.strip() == "[DONE]":
                        return
                    delta = json.loads(data)["choices"][0].get("delta", {})
                    if content := delta.get("content"):
                        yield content
```

```python
import hashlib


class FakeEmbedding:
    def __init__(self, dim: int = 4):
        self.dim = dim

    async def embed(self, texts: list[str]) -> list[list[float]]:
        out = []
        for t in texts:
            h = hashlib.sha256(t.encode()).digest()
            vec = [b / 255.0 for b in h[: self.dim]]
            out.append(vec)
        return out
```

```python
# app/providers/embedding/openai_compat.py
import httpx


class OpenAICompatEmbedding:
    def __init__(self, base_url: str, model: str, api_key: str | None = None, batch_size: int = 16,
                 timeout: float = 120.0):
        self.base_url = base_url.rstrip("/")
        self.model = model
        self.api_key = api_key
        self.batch_size = batch_size  # 对本地 oMLX 小批次，避免与对话争资源
        self.timeout = timeout

    async def embed(self, texts: list[str]) -> list[list[float]]:
        headers = {"Authorization": f"Bearer {self.api_key}"} if self.api_key else {}
        result: list[list[float]] = []
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            for i in range(0, len(texts), self.batch_size):
                batch = texts[i: i + self.batch_size]
                resp = await client.post(f"{self.base_url}/embeddings",
                                         json={"model": self.model, "input": batch}, headers=headers)
                resp.raise_for_status()
                data = sorted(resp.json()["data"], key=lambda d: d["index"])
                result.extend(d["embedding"] for d in data)
        return result
```

```python
# app/services/ingestion/chunking.py
import re

PUNCT = re.compile(r"(?<=[。！？.!?；;])\s*")


def _tokens(text: str) -> int:
    return max(1, len(text) // 2)


def split_blocks(blocks: list[dict], max_tokens: int = 500, overlap_ratio: float = 0.1) -> list[dict]:
    chunks: list[dict] = []
    buf: list[str] = []
    buf_tokens = 0
    meta = {"heading_path": "", "page_no": None}

    def flush():
        nonlocal buf, buf_tokens
        if buf:
            chunks.append({"text": "".join(buf), "heading_path": meta["heading_path"],
                           "page_no": meta["page_no"], "token_count": _tokens("".join(buf))})
            overlap_len = int(buf_tokens * overlap_ratio)
            kept: list[str] = []
            kept_tokens = 0
            while buf and kept_tokens < overlap_len:
                piece = buf.pop(0)
                kept.append(piece)
                kept_tokens += _tokens(piece)
            buf = kept
            buf_tokens = kept_tokens
        else:
            buf_tokens = 0

    for b in blocks:
        text, hp, page = b["text"], b["heading_path"], b["page_no"]
        if _tokens(text) > max_tokens:
            flush()
            for sent in PUNCT.split(text):
                if not sent:
                    continue
                if buf_tokens + _tokens(sent) > max_tokens and buf:
                    # 按句子硬切（保底：单句超限直接整句成块）
                    flush()
                # 携带该块的元数据
                meta = {"heading_path": hp, "page_no": page}
                buf.append(sent)
                buf_tokens += _tokens(sent)
            flush()
            continue
        if buf and (meta["heading_path"] != hp or meta["page_no"] != page):
            flush()
        meta = {"heading_path": hp, "page_no": page}
        buf.append(text)
        buf_tokens += _tokens(text)
        if buf_tokens >= max_tokens:
            flush()
    flush()
    return chunks
```

```python
# app/services/ingestion/parsing.py
from pathlib import Path

import fitz  # pymupdf
from docx import Document as DocxDocument
from markdown_it import MarkdownIt

md_parser = MarkdownIt()

SUPPORTED = {".md", ".txt", ".pdf", ".docx"}


def parse_file(path: Path, mime: str) -> list[dict]:
    ext = path.suffix.lower()
    if ext not in SUPPORTED:
        raise ValueError(f"不支持的文件类型: {ext}")
    return {".md": _parse_markdown, ".txt": _parse_text,
            ".pdf": _parse_pdf, ".docx": _parse_docx}[ext](path)


def _parse_markdown(path: Path) -> list[dict]:
    tokens = md_parser.parse(path.read_text(encoding="utf-8"))
    blocks, headings, buf = [], [], []

    def flush():
        text = "".join(buf).strip()
        if text:
            blocks.append({"text": text, "page_no": None, "heading_path": " > ".join(headings)})
        buf.clear()

    for tok in tokens:
        if tok.type == "heading_open":
            flush()
            level = int(tok.tag[1])
            del headings[level - 1:]
        elif tok.type == "inline":
            if tok.tag and tok.tag.startswith("h"):  # 标题文本
                headings.append(tok.content)
                flush_current_heading_done = True
            else:
                buf.append(tok.content + "\n\n")
    flush()
    return blocks
```

说明：markdown-it 的 inline token 无 `tag` 属性，实现时用「heading_open 后紧跟的 inline」判定标题文本（配对遍历 tokens），标题文本进 `headings` 而不进正文。`.txt` 按 `\n\s*\n` 分段；`.pdf` 用 `fitz.open` 逐页 `page.get_text()` 按空白分段并填 `page_no`；`.docx` 遍历 `doc.paragraphs`，`style.name.startswith("Heading")` 更新标题层级，普通段落为 Block。四个解析器都返回统一 Block 结构。

```python
async def ingest_document(document_id: int) -> None:
    with SessionLocal() as s:
        doc = s.get(Document, document_id)
        if not doc:
            return
        try:
            doc.status = "parsing"; s.commit()
            blocks = parse_file(Path(doc_file_path(doc)), doc.mime)
            chunks = split_blocks(blocks)
            doc.status = "embedding"; s.commit()
            emb_cfg = get_default_provider(s, "embedding")  # 无则 RuntimeError
            provider = build_embedding_provider(emb_cfg)    # fake|openai_compat
            # 先删旧向量（幂等重跑）
            s.execute(delete(ChunkEmbedding).where(ChunkEmbedding.document_id == document_id))
            s.execute(delete(Chunk).where(Chunk.document_id == document_id))
            s.add_all([Chunk(document_id=document_id, workspace_id=doc.workspace_id,
                             ordinal=i, content=c["text"], token_count=c["token_count"],
                             heading_path=c["heading_path"], page_no=c["page_no"])
                       for i, c in enumerate(chunks)])
            s.commit()
            db_chunks = s.execute(select(Chunk).where(Chunk.document_id == document_id)).scalars().all()
            vectors = await provider.embed([c.content for c in db_chunks])
            s.add_all([ChunkEmbedding(chunk_id=c.id, workspace_id=c.workspace_id,
                                      model_name=emb_cfg.model, dim=len(v), embedding=v)
                       for c, v in zip(db_chunks, vectors)])
            doc.status = "ready"; doc.error = None; s.commit()
        except Exception as e:
            s.rollback()
            doc.status = "failed"; doc.error = str(e)[:2000]; s.commit()
            raise
```

```python
        stuck = s.execute(select(Document).where(Document.status.in_(["parsing", "embedding", "pending"]))).scalars().all()
```

```ts
  | { type: "error"; message: string };

export async function* parseSSE(body: ReadableStream<Uint8Array>): AsyncGenerator<SSEEvent> {
  const reader = body.pipeThrough(new TextDecoderStream()).getReader();
  let buf = "";
  while (true) {
    const { done, value } = await reader.read();
    if (done) return;
    buf += value;
    const parts = buf.split("\n\n");
    buf = parts.pop() ?? "";
    for (const p of parts) {
      const line = p.split("\n").find((l) => l.startsWith("data: "));
      if (line) yield JSON.parse(line.slice(6));
    }
  }
}
```
